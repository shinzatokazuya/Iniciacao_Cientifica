CREATE TABLE locais (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    cidade TEXT,
    UF TEXT,
    regiao TEXT
);

CREATE TABLE clubes (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT,
    local_id INTEGER,
    FOREIGN KEY (local_id) REFERENCES locais(ID)
);

CREATE TABLE campeonatos (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT,
    pais TEXT,
    entidade TEXT
);

CREATE TABLE edicoes (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    campeonato_id INTEGER,
    ano INTEGER,
    FOREIGN KEY (campeonato_id) REFERENCES campeonatos(ID)
);

CREATE TABLE partidas (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    edicao_id INTEGER,
    estadio_id INTEGER,
    data_hora DATETIME,
    fase TEXT,
    mandante_id INTEGER,
    visitante_id INTEGER,
    mandante_placar INTEGER,
    visitante_placar INTEGER,
    mandante_penalti INTEGER,
    visitante_penalti INTEGER,
    prorrogacao INTEGER DEFAULT 0,
    FOREIGN KEY (mandante_id) REFERENCES clubes(ID),
    FOREIGN KEY (visitante_id) REFERENCES clubes(ID),
    FOREIGN KEY (edicao_id) REFERENCES edicoes(ID),
    FOREIGN KEY (estadio_id) REFERENCES estadios(ID)
);

CREATE TABLE estadios (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT,
    capacidade INTEGER,
    local_id INTEGER,
    FOREIGN KEY (local_id) REFERENCES locais(ID)
);

CREATE TABLE jogadores (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    nome TEXT,
    nascimento DATE,
    nacionalidade TEXT,
    clube_id INTEGER,
    FOREIGN KEY (clube_id) REFERENCES clubes(ID)
);

CREATE TABLE jogadores_em_partida (
    partida_id INTEGER,
    jogador_id INTEGER,
    titular BOOLEAN,
    minutos_jogados INTEGER,
    gols INTEGER,
    assistencias INTEGER,
    PRIMARY KEY (partida_id, jogador_id),
    FOREIGN KEY (partida_id) REFERENCES partidas(ID),
    FOREIGN KEY (jogador_id) REFERENCES jogadores(ID)
);
